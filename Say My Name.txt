"use client";

import React, { useEffect, useRef, useState } from "react";

const TOTAL_FRAMES = 240;
const BASE_PATH = process.env.NEXT_PUBLIC_BASE_PATH || "";

export default function CinematicVideo() {
  const canvasRef = useRef<HTMLCanvasElement | null>(null);
  const containerRef = useRef<HTMLDivElement | null>(null);
  const glowRef = useRef<HTMLDivElement | null>(null);
  const progressBarRef = useRef<HTMLDivElement | null>(null);

  const framesRef = useRef<ImageBitmap[]>([]);
  const targetFrameRef = useRef<number>(0);
  const currentFrameRef = useRef<number>(0);
  const rafId = useRef<number>(0);

  const targetTilt = useRef<{ x: number; y: number }>({ x: 0, y: 0 });
  const currentTilt = useRef<{ x: number; y: number }>({ x: 0, y: 0 });

  const [loadingProgress, setLoadingProgress] = useState<number>(0);
  const [isLoaded, setIsLoaded] = useState<boolean>(false);

  // 1. Asynchronous Batch Preloading via ImageBitmap
  useEffect(() => {
    let active = true;

    async function loadAllFrames() {
      const bitmaps: ImageBitmap[] = [];
      let loadedCount = 0;

      // Batch download in chunks of 12 parallel requests
      const BATCH_SIZE = 12;
      for (let i = 1; i <= TOTAL_FRAMES; i += BATCH_SIZE) {
        if (!active) return;
        const batchIndices: number[] = [];
        for (let j = i; j < i + BATCH_SIZE && j <= TOTAL_FRAMES; j++) {
          batchIndices.push(j);
        }

        const batchPromises = batchIndices.map(async (idx) => {
          const frameNum = String(idx).padStart(4, "0");
          const url = `${BASE_PATH}/frames/frame_${frameNum}.webp`;

          try {
            const res = await fetch(url);
            const blob = await res.blob();
            const bitmap = await createImageBitmap(blob);
            bitmaps[idx - 1] = bitmap;
          } catch {
            // Fallback or retry logic if needed
          } finally {
            loadedCount++;
            if (active) {
              setLoadingProgress(Math.floor((loadedCount / TOTAL_FRAMES) * 100));
            }
          }
        });

        await Promise.all(batchPromises);
      }

      if (active) {
        framesRef.current = bitmaps;
        setIsLoaded(true);
      }
    }

    loadAllFrames();

    return () => {
      active = false;
      framesRef.current.forEach((bitmap) => bitmap.close());
    };
  }, []);

  // 2. Cursor, Scroll & 3D Tilt Binding
  useEffect(() => {
    const handleMouseMove = (e: MouseEvent) => {
      const { innerWidth, innerHeight } = window;
      const normX = e.clientX / innerWidth;
      const normY = e.clientY / innerHeight;

      // Parallax target delta: -1.0 to 1.0
      targetTilt.current = {
        x: (e.clientX - innerWidth / 2) / (innerWidth / 2),
        y: (e.clientY - innerHeight / 2) / (innerHeight / 2),
      };

      // Spot glow tracking
      if (glowRef.current) {
        glowRef.current.style.background = `radial-gradient(circle 650px at ${(normX * 100).toFixed(2)}% ${(normY * 100).toFixed(2)}%, rgba(220, 20, 60, 0.22), transparent 70%)`;
      }

      updateTargetFrame(normX);
    };

    const handleScroll = () => {
      updateTargetFrame(0.5);
    };

    const updateTargetFrame = (cursorXRatio: number) => {
      const scrollY = window.scrollY || window.pageYOffset;
      const maxScroll = Math.max(1, document.documentElement.scrollHeight - window.innerHeight);
      const scrollRatio = Math.min(1, Math.max(0, scrollY / maxScroll));

      if (progressBarRef.current) {
        progressBarRef.current.style.transform = `scaleX(${scrollRatio})`;
      }

      // Hybrid calculation: 70% scroll, 30% horizontal cursor drift
      const factor = Math.min(1, Math.max(0, scrollRatio * 0.7 + cursorXRatio * 0.3));
      targetFrameRef.current = Math.min(TOTAL_FRAMES - 1, Math.max(0, factor * (TOTAL_FRAMES - 1)));
    };

    window.addEventListener("mousemove", handleMouseMove, { passive: true });
    window.addEventListener("scroll", handleScroll, { passive: true });

    return () => {
      window.removeEventListener("mousemove", handleMouseMove);
      window.removeEventListener("scroll", handleScroll);
    };
  }, []);

  // 3. High-Frequency Canvas RAF Render Loop
  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d", { alpha: false });
    if (!ctx) return;

    const render = () => {
      // Smooth frame interpolation (LERP)
      currentFrameRef.current += (targetFrameRef.current - currentFrameRef.current) * 0.15;
      const activeIdx = Math.round(currentFrameRef.current);

      const bitmap = framesRef.current[activeIdx];
      if (bitmap && ctx) {
        // Draw frame with crisp canvas dimensions
        if (canvas.width !== bitmap.width || canvas.height !== bitmap.height) {
          canvas.width = bitmap.width;
          canvas.height = bitmap.height;
        }
        ctx.drawImage(bitmap, 0, 0);
      }

      // 3D Parallax Tilt LERP
      currentTilt.current.x += (targetTilt.current.x - currentTilt.current.x) * 0.08;
      currentTilt.current.y += (targetTilt.current.y - currentTilt.current.y) * 0.08;

      if (containerRef.current) {
        const dx = currentTilt.current.x;
        const dy = currentTilt.current.y;
        containerRef.current.style.transform = `scale(1.06) translate3d(${dx * -15}px, ${dy * -15}px, 0) rotateX(${dy * -2}deg) rotateY(${dx * 2}deg)`;
      }

      rafId.current = requestAnimationFrame(render);
    };

    rafId.current = requestAnimationFrame(render);

    return () => cancelAnimationFrame(rafId.current);
  }, []);

  return (
    <>
      <div id="cine-progress" ref={progressBarRef} style={{ width: "100%" }} />

      {/* Cyber Preloader Bar */}
      {!isLoaded && (
        <div className="fixed top-0 left-0 right-0 z-50 h-1 bg-neutral-900">
          <div
            className="h-full bg-red-600 transition-all duration-100 ease-out shadow-[0_0_8px_#ff1a3d]"
            style={{ width: `${loadingProgress}%` }}
          />
        </div>
      )}

      <div className="fixed inset-0 pointer-events-none z-0 overflow-hidden bg-black">
        <div
          ref={containerRef}
          className="relative w-full h-full will-change-transform transition-transform duration-75 ease-out"
        >
          <canvas
            ref={canvasRef}
            className="w-full h-full object-cover object-center filter contrast-125 brightness-95"
          />
        </div>

        {/* Ambient Film Overlays */}
        <div className="cine-vignette" />
        <div id="cine-glow" ref={glowRef} className="cine-glow" />
        <div className="cine-scan" />
        <div className="cine-grain" />
      </div>
    </>
  );
}
